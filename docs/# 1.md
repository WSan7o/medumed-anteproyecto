# 1. Crear la carpeta local y entrar a ella
mkdir medumed-anteproyecto
cd medumed-anteproyecto

# 2. Inicializar Git y crear la estructura de carpetas
git init
mkdir docs forms resources
touch README.md .gitignore docs/anteproyecto.md

# 3. Guardar los archivos locales
git add .
git commit -m "feat: estructura inicial del anteproyecto Medumed"

# 4. Vincular con GitHub y subir el contenido
git branch -M main
git remote add origin https://github.com/WSan7o/medumed-anteproyecto.git
git push -u origin main
